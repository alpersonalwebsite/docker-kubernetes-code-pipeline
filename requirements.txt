#     from jinja2 import escape
#   ImportError: cannot import name 'escape' from 'jinja2'
#
# jinja2 3.0 removed the `escape` re-export that flask 1.1.2 imports at module scope, and
# Werkzeug 2.x moved several names flask 1.x expects. So `pip install -r requirements.txt`
